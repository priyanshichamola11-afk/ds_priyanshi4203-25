{

    {
        printf("Enter element %d: ", i + 1);
        scanf("%d", &arr[i]);

#include <stdio.h>

char str[50];
char key;

int linearSearch(int index)
{
    if (str[index] == '\0')
    {
        return -1;
    }

    if (str[index] == key)
    {
        return index;
    }

    return linearSearch(index + 1);
}

int main()
{
    int result;

    printf("Enter a string: ");
    scanf("%s", str);

    printf("Enter the character to search: ");
    scanf(" %c", &key);

    result = linearSearch(0);

    if (result == -1)
    {
        printf("'%c' is not found in the string.\n", key);
    }
    else
    {
        printf("'%c' is found at index %d (position %d).\n", key, result, result + 1);
    }

    return 0;
}

  1. ENTER A INTEGER ARRAY FROM THE USER, ENTER AN INTEGER VALUE AND PERFORM BINARY SEARCH ALGORITHM THROUGH ITERATION.

#include <stdio.h>

int main()
{
    int arr[50];
    int n, i, key;
    int low, high, mid;
    int found = 0;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    printf("Enter the elements in ascending (sorted) order:\n");
    for (i = 0; i < n; i++)
    {
        printf("Enter element %d: ", i + 1);
        scanf("%d", &arr[i]);
    }

    printf("Enter the value to search: ");
    scanf("%d", &key);

    low = 0;
    high = n - 1;

    while (low <= high)
    {
        mid = (low + high) / 2;

        if (arr[mid] == key)
        {
            printf("%d is found at index %d (position %d).\n", key, mid, mid + 1);
            found = 1;
            break;
        }
        else if (arr[mid] < key)
        {
            low = mid + 1;
        }
        else
        {
            high = mid - 1;
        }
    }

    if (found == 0)
    {
        printf("%d is not found in the array.\n", key);
    }

    return 0;
}

#include <stdio.h>
#include <string.h>

char str[50];
char key;

int binarySearch(int low, int high)
{
    int mid;

    if (low > high)
    {
        return -1;
    }

    mid = (low + high) / 2;

    if (str[mid] == key)
    {
        return mid;
    }
    else if (str[mid] < key)
    {
        return binarySearch(mid + 1, high);
    }
    else
    {
        return binarySearch(low, mid - 1);
    }
}

int main()
{
    int len, result;

    printf("Enter a sorted string (ascending order, no spaces): ");
    scanf("%s", str);

    printf("Enter the character to search: ");
    scanf(" %c", &key);

    len = strlen(str);

    result = binarySearch(0, len - 1);

    if (result == -1)
    {
        printf("'%c' is not found in the string.\n", key);
    }
    else
    {
        printf("'%c' is found at index %d (position %d).\n", key, result, result + 1);
    }

    return 0;
}
