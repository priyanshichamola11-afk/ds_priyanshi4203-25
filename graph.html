  1. ENTER THE VERTICES AND EDGES AND REPRESENT IN ADJACENCY MATRIX AND PRINT IT..

    2. 
#include <stdio.h>

int matrix[10][10];

int main()
{
    int v, e, i, j, a, b;

    printf("Enter the number of vertices: ");
    scanf("%d", &v);

    printf("Enter the number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < v; i++)
    {
        for (j = 0; j < v; j++)
        {
            matrix[i][j] = 0;
        }
    }

    for (i = 0; i < e; i++)
    {
        printf("Enter edge %d (two vertices, 0 to %d): ", i + 1, v - 1);
        scanf("%d %d", &a, &b);

        if (a < 0 || a >= v || b < 0 || b >= v)
        {
            printf("Invalid vertex! Enter this edge again.\n");
            i--;
        }
        else
        {
            matrix[a][b] = 1;
            matrix[b][a] = 1;
        }
    }

    printf("\nAdjacency Matrix:\n");

    printf("   ");
    for (i = 0; i < v; i++)
    {
        printf("%d ", i);
    }
    printf("\n");

    for (i = 0; i < v; i++)
    {
        printf("%d  ", i);
        for (j = 0; j < v; j++)
        {
            printf("%d ", matrix[i][j]);
        }
        printf("\n");
    }

    return 0;
}

  1. TRAVERSE THE ADJACENT MATRIX WITH BREADTH FIRST SEARCH.

#include <stdio.h>

int matrix[10][10];
int visited[10];
int v;

int queue[10];
int front = 0;
int rear = -1;

int isEmpty()
{
    if (front > rear)
    {
        return 1;
    }
    else
    {
        return 0;
    }
}

void insert(int x)
{
    rear++;
    queue[rear] = x;
}

int delete_element()
{
    int x = queue[front];
    front++;
    return x;
}

void bfs(int start)
{
    int curr, i;

    visited[start] = 1;
    insert(start);

    while (isEmpty() == 0)
    {
        curr = delete_element();
        printf("%d ", curr);

        for (i = 0; i < v; i++)
        {
            if (matrix[curr][i] == 1 && visited[i] == 0)
            {
                visited[i] = 1;
                insert(i);
            }
        }
    }
}
