struct Node *last = NULL;


// ISEMPTY

    return (last == NULL);
}

// ISFULL (list is full when malloc fails)
int isFull() {
    struct Node *temp = (struct Node*)malloc(sizeof(struct Node));
    if (temp == NULL) return 1;
    free(temp);
    return 0;
}


// INSERT AT BEGIN
void insertAtBegin(char value[]) {
    struct Node *newNode = (struct Node*)malloc(sizeof(struct Node));
    if (newNode == NULL) {
        printf("List is full\n");
        return;
    }
    strcpy(newNode->data, value);

    if (last == NULL) {
        last = newNode;
        newNode->next = newNode;        // points to itself
    } else {
        newNode->next = last->next;     // new node -> old first node
        last->next = newNode;           // last node -> new node
    }
    printf("Inserted at beginning\n");
}

// INSERT AT END
void insertAtEnd(char value[]) {
    struct Node *newNode = (struct Node*)malloc(sizeof(struct Node));
    if (newNode == NULL) {
        printf("List is full\n");
        return;
    }
    strcpy(newNode->data, value);

    if (last == NULL) {
        last = newNode;
        newNode->next = newNode;        // points to itself
    } else {
        newNode->next = last->next;     // new node -> first node
        last->next = newNode;           // old last node -> new node
        last = newNode;                 // new node is now the last node
    }
    printf("Inserted at end\n");
}

// INSERT AT POSITION (position starts from 1)
void insertAtPosition(char value[], int position) {
    int i;

    if (position < 1) {
        printf("Invalid position\n");
        return;
    }
    if (position == 1) {
        insertAtBegin(value);
        return;
    }
    if (last == NULL) {
        printf("List is empty, only position 1 is valid\n");
        return;
    }

    // move to the node just before the given position
    struct Node *p = last->next;
    for (i = 1; i < position - 1; i++) {
        if (p == last) {                // reached the end of the list
            printf("Invalid position\n");
            return;
        }
        p = p->next;
    }

    struct Node *newNode = (struct Node*)malloc(sizeof(struct Node));
    if (newNode == NULL) {
        printf("List is full\n");
        return;
    }
    strcpy(newNode->data, value);

    newNode->next = p->next;            // new node -> next node
    p->next = newNode;                  // previous node -> new node
    if (p == last) {
        last = newNode;                 // inserted at end, so update last
    }
    printf("Inserted at position %d\n", position);
}


// DELETE AT BEGIN
void deleteAtBegin() {
    if (last == NULL) {
        printf("List is empty\n");
        return;
    }
    struct Node *first = last->next;

    if (first == last) {
        last = NULL;                    // only one node was present
    } else {
        last->next = first->next;       // last node -> second node
    }
    printf("Deleted: %s\n", first->data);
    free(first);
}

// DELETE AT END
void deleteAtEnd() {
    if (last == NULL) {
        printf("List is empty\n");
        return;
    }
    if (last->next == last) {           // only one node
        printf("Deleted: %s\n", last->data);
        free(last);
        last = NULL;
        return;
    }

    // find the second last node
    struct Node *p = last->next;
    while (p->next != last) {
        p = p->next;
    }
    p->next = last->next;               // second last -> first node
    printf("Deleted: %s\n", last->data);
    free(last);
    last = p;                           // second last is the new last
}

// DELETE AT POSITION
void deleteAtPosition(int position) {
    int i;

    if (last == NULL) {
        printf("List is empty\n");
        return;
    }
    if (position < 1) {
        printf("Invalid position\n");
        return;
    }
    if (position == 1) {
        deleteAtBegin();
        return;
    }

    // move to the node just before the given position
    struct Node *p = last->next;
    for (i = 1; i < position - 1; i++) {
        if (p == last) {
            printf("Invalid position\n");
            return;
        }
        p = p->next;
    }
    if (p == last) {                    // no node exists after this one
        printf("Invalid position\n");
        return;
    }

    struct Node *temp = p->next;        // node to be deleted
    p->next = temp->next;               // skip the deleted node
    if (temp == last) {
        last = p;                       // deleted the last node, so update last
    }
    printf("Deleted: %s\n", temp->data);
    free(temp);
}

#include <stdio.h>

struct Node
{
    int coeff;
    int exp;
    struct Node *next;

struct Node *insertAtEnd(struct Node *head, int coeff, int exp)
{
    struct Node *newNode = (struct Node *)malloc(sizeof(struct Node));
    struct Node *temp;

    newNode->coeff = coeff;
    newNode->exp = exp;
    newNode->next = NULL;

    if (head == NULL)
    {
        return newNode;
    }

    temp = head;
    while (temp->next != NULL)
    {
        temp = temp->next;
    }
    temp->next = newNode;
    return head;
}

// Read a polynomial from the user
struct Node *readPolynomial()
{
    struct Node *head = NULL;
    int n, i, coeff, exp;

    printf("Enter number of terms: ");
    scanf("%d", &n);

    printf("Enter terms in DECREASING order of exponent\n");
    for (i = 1; i <= n; i++)
    {
        printf("Term %d - Enter coefficient and exponent: ", i);
        scanf("%d %d", &coeff, &exp);
        head = insertAtEnd(head, coeff, exp);
    }
    return head;
}


struct Node *addPolynomials(struct Node *p1, struct Node *p2)
{
    struct Node *result = NULL;
    int sum;

    while (p1 != NULL && p2 != NULL)
    {
        if (p1->exp == p2->exp)
        { // same exponent: add coefficients
            sum = p1->coeff + p2->coeff;
            if (sum != 0)
            {
                result = insertAtEnd(result, sum, p1->exp);
            }
            p1 = p1->next;
            p2 = p2->next;
        }
        else if (p1->exp > p2->exp)
        { // poly1 has bigger exponent
            result = insertAtEnd(result, p1->coeff, p1->exp);
            p1 = p1->next;
        }
        else
        { // poly2 has bigger exponent
            result = insertAtEnd(result, p2->coeff, p2->exp);
            p2 = p2->next;
        }
    }

    // copy remaining terms of poly1 (if any)
    while (p1 != NULL)
    {
        result = insertAtEnd(result, p1->coeff, p1->exp);
        p1 = p1->next;
    }

    // copy remaining terms of poly2 (if any)
    while (p2 != NULL)
    {
        result = insertAtEnd(result, p2->coeff, p2->exp);
        p2 = p2->next;
    }

    return result;
}

// Display a polynomial
void display(struct Node *head)
{

        if (head->next != NULL)
        {
            printf(" + ");
        }
        head = head->next;
    }
    printf("\n");
}
