    2. 

#include <stdio.h>

void FIBONACCI(int n)
{
    int a, b, res;
    a = 0;
    b = 1;
    printf("%d %d", a, b);

    for (int i = 0; i < n - 2; i++)
    {
        res = a + b;
        printf(" %d", res);
        a = b;
        b = res;
    }
}

int main()
{
    int n;
    printf("Enter how many terms you want in your Fibonacci series: ");
    scanf("%d", &n);

    FIBONACCI(n);
}

void binary(int n) {

}

    else

#include <stdio.h>

int sumDigits(int n) {
    if (n == 0)
        return 0;                       
    return (n % 10) + sumDigits(n / 10);
}

int main() {
    int num;

    printf("Enter a number: ");
    scanf("%d", &num);

    if (num < 0){
        num = -num;                     
    }
    printf("Sum of digits = %d", sumDigits(num));
    return 0;
}

  1. FIND THE ARITHMETIC MEAN OF AN INTEGER ARRAY USING RECURSION. 

    2. 
    #include <stdio.h>

int sum(int a[], int n)
{
    if (n == 0)
        return 0;
        
    return a[n - 1] + sum(a, n - 1);
}

int main()
{
    int arr[100], n, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter %d values: ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    printf("Mean = %.2f", (double)sum(arr, n) / n);
    return 0;
}
