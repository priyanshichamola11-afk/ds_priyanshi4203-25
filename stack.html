#include <stdio.h>

int stack[5];
int top = -1;
 int SIZE = 5;

int isEmpty()
{
    if (top == -1)
    {
        return 1;
    }
    else
    {
        return 0;
    }
}

int isFull()
{
    if (top == SIZE - 1)
    {
        return 1;
    }
    else
    {
        return 0;
    }
}

void push(int x)
{
    if (isFull())
    {
        printf("Stack Overflow! The stack is full.\n");
        return;
    }
    top++;
    stack[top] = x;
    printf("%d pushed.\n", x);
}

void pop()
{
    if (isEmpty())
    {
        printf("Stack Underflow! The stack is empty.\n");
        return;
    }
    printf("%d popped.\n", stack[top]);
    top--;
}

void peek()
{
    if (isEmpty())
    {
        printf("The stack is empty.\n");
        return;
    }
    printf("Top element: %d\n", stack[top]);
}

{
    int choice, value;

    while (1)
    {
        printf("\nSTACK MENU\n");
        printf("1. Push\n");

        }

            peek();
        }

            {
                printf("No, the stack is not empty.\n");

{

{
    char str[50];
    int i, len;
    int error = 0;

    printf("Enter an expression: ");
    fgets(str, 50, stdin);

    len = strlen(str);

    for (i = 0; i < len; i++)
    {
        if (str[i] == '(')
        {
            push(str[i]);
        }
        else if (str[i] == ')')
        {
            if (top == -1)
            {
                error = 1;
                break;
            }
            else
            {
                pop();
            }
        }
    }

    if (error == 0 && top == -1)
    {
        printf("Brackets are balanced.\n");
    }
    else
    {
        printf("Brackets are NOT balanced.\n");
    }

    return 0;
}

  1. SORT THE STACK IN ASCENDING ORDER USING INTEGER ARRAY. 

    2. 
#include <stdio.h>

int stack[50];
int top = -1;

int temp[50];
int topT = -1;

void push(int x)
{
    top++;
    stack[top] = x;
}

int pop()
{
    int x = stack[top];
    top--;
    return x;
}

void pushT(int x)
{
    topT++;
    temp[topT] = x;
}

int popT()
{
    int x = temp[topT];
    topT--;
    return x;
}

void sortStack()
{
    int x, i;

    while (top != -1)
    {
        x = pop();

        while (topT != -1 && temp[topT] > x)
        {
            push(popT());
        }

        pushT(x);
    }


    for (i = 0; i <= topT; i++)
    {
        stack[i] = temp[i];
    }
    top = topT;
    topT = -1;
}

int main()
{
    int n, i, value;

    printf("How many elements? ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
    {
        printf("Enter element %d: ", i + 1);
        scanf("%d", &value);
        push(value);
    }

    sortStack();

    printf("Sorted stack (bottom to top): ");
    for (i = 0; i <= top; i++)
    {
        printf("%d ", stack[i]);
    }
    printf("\n");

    return 0;
}
