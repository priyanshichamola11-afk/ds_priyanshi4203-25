struct Node *buildTree(int inStart, int inEnd)
{
    struct Node *root;
    int inIndex;

    if (inStart > inEnd)
    {
        return NULL;
    }

    root = (struct Node *)malloc(sizeof(struct Node));
    root->data = postorder[postIndex];
    root->left = NULL;
    root->right = NULL;
    postIndex--;

    if (inStart == inEnd)
    {
        return root;
    }

    inIndex = search(inStart, inEnd, root->data);

    root->right = buildTree(inIndex + 1, inEnd);
    root->left = buildTree(inStart, inIndex - 1);

    return root;
}

int maxDepth(struct Node *root)
{
    int leftDepth, rightDepth;

    if (root == NULL)
    {
        return 0;

    leftDepth = maxDepth(root->left);
    rightDepth = maxDepth(root->right);

int tree[31];
int temp[31];
int tempCount = 0;

void clearTree()
{
    int i;

    for (i = 0; i < 31; i++)
    {
        tree[i] = -1;
    }
}

// return 1 = inserted, 0 = duplicate, -1 = no space
int insertSilent(int x)
{
    int i = 0;

    while (i < 31)
    {
        if (tree[i] == -1)
        {
            tree[i] = x;
            return 1;
        }

        if (x == tree[i])
        {
            return 0;
        }

        if (x < tree[i])
        {
            i = 2 * i + 1;
        }
        else
        {
            i = 2 * i + 2;
        }
    }

    return -1;
}

void insert(int x)
{
    int result = insertSilent(x);

    if (result == 1)
    {
        printf("%d inserted.\n", x);
    }
    else if (result == 0)
    {
        printf("%d already exists.\n", x);
    }
    else
    {
        printf("Tree Overflow! No space for %d.\n", x);
    }
}

int search(int x)
{
    int i = 0;

    while (i < 31 && tree[i] != -1)
    {
        if (x == tree[i])
        {
            return 1;
        }

        if (x < tree[i])
        {
            i = 2 * i + 1;
        }
        else
        {
            i = 2 * i + 2;
        }
    }

    return 0;
}

void collect(int i, int del)
{
    if (i >= 31 || tree[i] == -1)
    {
        return;
    }

    if (tree[i] != del)
    {
        temp[tempCount] = tree[i];
        tempCount++;
    }

    collect(2 * i + 1, del);
    collect(2 * i + 2, del);
}

void delete_element(int x)
{
    int i;

    if (search(x) == 0)
    {
        printf("%d not found.\n", x);
        return;
    }

    tempCount = 0;
    collect(0, x);
    clearTree();

    for (i = 0; i < tempCount; i++)
    {
        insertSilent(temp[i]);
    }

    printf("%d deleted.\n", x);
}

void preorder(int i)
{
    if (i >= 31 || tree[i] == -1)

        return;
