#include<string.h>
void swap(char *a,char *b){
    char t=*a;
    *a=*b;
    *b=t;
}
void perm(char s[],int l,int r){
    int i;
    if(l==r){
        printf("%s\n",s);
    }
    else{
        for(i=l;i<=r;i++){
            swap(&s[l],&s[i]);
            perm(s,l+1,r);
            swap(&s[l],&s[i]);
        }
    }
}

    char s[50];

    gets(s);
    perm(s,0,strlen(s)-1);

}

#include<string.h>
struct dict{
    char w[50],m[100];
};
struct dict d[10];
int n=0;

void create(){
    int i;
    if(n==10){
        printf("dictionary is full");
    }
    else{
        printf("enter word: ");
        scanf("%49s",d[n].w);
        printf("enter meaning: ");
        scanf(" %99[^\n]",d[n].m);
        n++;
        printf("word added");
    }
}
void delete(){
    char x[50];
    int i,j,f=0;
    printf("enter word to delete: ");
    scanf("%49s",x);
    for(i=0;i<n;i++){
        if(strcmp(d[i].w,x)==0){
            for(j=i;j<n-1;j++){
                d[j]=d[j+1];
            }
            n--;
            f=1;
            break;
        }
    }
    if(f==1){
        printf("word deleted");
    }
    else{
        printf("word not found");
    }
}

    char x[50];
    int i,f=0;
    printf("enter word to search: ");
    scanf("%49s",x);

            printf("meaning: %s",d[i].m);
            f=1;
            break;
        }
    }
    if(f==0){
        printf("word not found");
    }
}

    int x,i;
