    {
        return 1;

    else
    {
        return 0;
    }

{
    if (isFull())

        printf("The queue is empty.\n");
        return;

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *next;
};

struct Node *front = NULL;
struct Node *rear = NULL;

int isEmpty()
{
    if (front == NULL)
    {
        return 1;
    }
    else
    {
        return 0;
    }
}

int isFull()
{
    struct Node *test = (struct Node *)malloc(sizeof(struct Node));

    if (test == NULL)
    {
        return 1;
    }
    else
    {
        free(test);
        return 0;
    }
}

void insert(int value)
{
    struct Node *newNode;

    if (isFull())
    {
        printf("Queue Overflow! Memory is full.\n");
        return;
    }

    newNode = (struct Node *)malloc(sizeof(struct Node));
    newNode->data = value;
    newNode->next = NULL;

    if (front == NULL)
    {
        front = newNode;
        rear = newNode;
    }
    else
    {
        rear->next = newNode;
        rear = newNode;
    }

    printf("%d inserted.\n", value);
}

void delete_element()
{
    struct Node *temp;

    if (isEmpty())
    {
        printf("Queue Underflow! The queue is empty.\n");
        return;
    }

    temp = front;
    printf("%d deleted.\n", temp->data);
    front = front->next;

    if (front == NULL)
    {
        rear = NULL;
    }

    free(temp);
}

void peek()

    {
        printf("The queue is empty.\n");
